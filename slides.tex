% SLIDE 2: Motivation
%================================================
\begin{frame}{Motivation: the data fusion dilemma}
    \vspace{0.05cm}
    \textbf{The clinical trade-off}:
    \begin{itemize}
        \setlength{\itemsep}{0.12cm}
        \item \emph{Randomised controlled trials (RCTs)}:
        \begin{itemize}
            \item Gold standard for internal validity through design;
            \item Small sample sizes, high costs, restrictive criteria: limited statistical power and poor generalisability.
        \end{itemize}
        \item \emph{Observational real-world data (RWD)}:
        \begin{itemize}
            \item Large patient cohorts, clinical registries, routine practice records;
            \item Vulnerable to unmeasured confounding and selection bias.
        \end{itemize}
    \end{itemize}

    \vspace{0.15cm}
    \begin{alertblock}{Objective of data fusion}
        Borrow statistical power from large observational cohorts while preserving the internal validity of the randomized trial.
    \end{alertblock}

    \vspace{0.1cm}
    The Bayesian paradigm offers a natural framework to introduce auxiliary evidence into priors and adaptively discount it under data conflict.
\end{frame}

%================================================
% SLIDE 3: Baseline Power Prior
%================================================
\begin{frame}{Baseline: the power prior (Ibrahim \& Chen, 2000)}
    \vspace{0.1cm}
    Let $D_0 = (n_0, y_0, X_0)$ denote historical data with likelihood $L(\theta \mid D_0)$, and $\pi_0(\theta \mid c_0)$ an initial prior indexed by $c_0$.

    \begin{block}{Power prior and posterior formulation}
    For a discounting hyperparameter $a_0 \in [0, 1]$:
    \begin{equation*}
        \pi(\theta \mid D_0, a_0) \propto L(\theta \mid D_0)^{a_0} \pi_0(\theta \mid c_0), \quad
        \pi(\theta \mid D_e, D_0, a_0) \propto L(\theta \mid D_e) \pi(\theta \mid D_0, a_0)
    \end{equation*}
    \end{block}

    \begin{itemize}
        \item $a_0 = 1$: unpenalized pooling; $a_0 = 0$ excludes external data entirely.
        \item \textit{Precision scaling:} Disperses historical prior concentration.
    \end{itemize}

    \vspace{0.15cm}
    \textbf{Limitations}:
    \begin{itemize}
        \setlength{\itemsep}{0.1cm}
        \item \emph{Fixed $a_0$}: Subjective elicitation; no internal commensurability check;
        \item \emph{Random $a_0$}: Naive joint prior $\pi(\theta, a_0 \mid D_0) \propto L^{a_0} \pi_0 \pi(a_0)$ violates the Likelihood Principle (scaling L changes posterior on $a_0$) and ignores $C(a_0)$.
    \end{itemize}
\end{frame}

%================================================
% SLIDE 4: Modified Power Prior
%================================================
\begin{frame}{The modified power prior}
    \begin{block}{Proper power prior (Duan et al., 2006; Neuenschwander et al., 2009)}
    Retaining $C(a_0) = [ \int_\Theta L(\theta \mid D_0)^{a_0} \pi_0(\theta \mid c_0) d\theta ]^{-1}$ restores probability calculus:
    \begin{equation*}
        \pi(\theta, a_0 \mid D_0) = C(a_0) L(\theta \mid D_0)^{a_0} \pi_0(\theta \mid c_0) \pi(a_0) \quad (a_0 \in A)
    \end{equation*}
    \end{block}

    \begin{itemize}
        \item \emph{Prevents collapse}: Without $C(a_0)$, $\pi(a_0 \mid \text{data})$ collapses to 0 even for identical datasets.
        \item \emph{Dynamic commensurability}:  Given $\pi(a_0 | D_e, D_0) \propto \pi(a_0)m(D_e | D_0, a_0)$, concordance pushes $a_0 \to 1$; conflict forces $a_0 \to 0$.
        \item \emph{*Multi-study}: Extends to vector weights $a_0 = (a_{01}, \dots, a_{0H})^\T$ for $H \ge 2$.
    \end{itemize}

    \begin{highlightbox}
        \textbf{\textcolor{MyRed}{\small The single-source bottleneck ($H = 1$)}}\par\vspace{2pt}
        {\footnotesize With only one historical study ($H=1$), data cannot reliably separate between-study heterogeneity from noise: $\pi(a_0 \mid \text{data})$ remains sensitive to the chosen $\pi(a_0)$.}
    \end{highlightbox}
\end{frame}

%================================================
% SLIDE 5: MAP Priors (Schmidli et al., 2014)
%================================================
\begin{frame}{Meta-analytic-predictive (MAP) priors}
    \vspace{-0.25cm}
    \textbf{Schmidli et al. (2014)}: Shift to hierarchical predictive borrowing for control response $\theta = g(\psi)$ across $H \ge 3$ historical studies to downsize concurrent controls.

    \begin{block}{Hierarchical exchangeability model}
    \begin{equation*}
        \theta_1, \dots, \theta_H, \theta_* \mid \mu, \tau \overset{\text{i.i.d.}}{\sim} \text{N}(\mu, \tau^2), \quad \pi(\mu, \tau) = \pi(\mu)\pi(\tau)
    \end{equation*}
    where $\tau \ge 0$ quantifies between-trial heterogeneity (half-\textit{t} or half-Normal).
    \end{block}

    \vspace{-0.18cm}
    \begin{itemize}
        \setlength{\itemsep}{0.04cm}
        \item Small values of $\tau$ reflect corss-trial concordance and facilitate borrowing;
        \item \emph{Conjugate mixture approximation}: The empirical MCMC predictive density is approximated via EM by a $K$-component mixture ($K \in \{2, 3\}$):
        \begin{equation*}
            \hat\pi_{\text{MAP}}(\psi_*) = \sum_{k=1}^K w_k \pi_c(\psi_* \mid \xi_k), \quad \sum_{k=1}^K w_k = 1, \quad w_k > 0
        \end{equation*}
        Provides closed-form prospective updates, bypassing MCMC during trial analysis.
    \end{itemize}
\end{frame}

%================================================
% SLIDE 6: Robust MAP Priors & Posterior Updating
%================================================
\begin{frame}{Robust MAP priors and posterior updating}
    \vspace{-0.32cm}
    To protect against non-exchangeability, they construct a robust mixture:

    \begin{block}{Robust mixture prior (Schmidli et al. (2014))}
    Augment with vague conjugate component $\pi_V(\psi_*)$ weighted by $w_R \in [0.10, 0.20]$:
    \begin{equation*}
        \hat\pi_{\text{HR}}(\psi_*) = (1 - w_R) \sum_{k=1}^K w_k \pi_c(\psi_* \mid \xi_k) + w_R \pi_V(\psi_*)
    \end{equation*}
    \end{block}

    %\vspace{-0.18cm}
    \textbf{Predictive likelihoods and updated weights}:
    \begin{itemize}
        \setlength{\itemsep}{0.03cm}
        \item Marginal likelihoods: $m_k = \int_\Psi f(Y_* \mid \psi_*) \pi_c(\psi_* \mid \xi_k) d\psi_*, \; m_V = \int_\Psi f(Y_* \mid \psi_*) \pi_V(\psi_*) d\psi_*$.
        \item Updated weights: $\tilde{w}_R = \frac{w_R m_V}{w_R m_V + (1 - w_R)\sum w_j m_j}, \quad \tilde{w}_k = \frac{w_k m_k}{\sum w_j m_j}$.
        \item Posterior retains mixture form: $(1 - \tilde{w}_R)\sum \tilde{w}_k \pi_c(\psi_* \mid \xi_k, Y_*) + \tilde{w}_R \pi_V(\psi_* \mid Y_*)$.
        \item $m_k/m_V$ acts as Bayes factor; conflict ($m_k \ll m_V$) drives $\tilde{w}_R \to 1$, automatically discounting historical controls.
    \end{itemize}
\end{frame}

%================================================
% SLIDE 7: Pairwise Shrinkage for H=1
%================================================
\begin{frame}{Pairwise shrinkage for $H = 1$ (R\"over \& Friede, 2020)}
    \begin{itemize}
    \item the MAP formulation rewuires $H \ge 3$ to reliably estimate heterogeneity $\tau$
    \item When $H=1$, frequentist random-effects often yield $\hat\tau^2 = 0$, underestimating variance uncertainty; moreover, symmetric exchangeability around a mean $\mu$ is hard to justify when combining RCT with RWD.
    \end{itemize}

    \begin{block}{Asymmetric reference model ($\theta_e = \mu$)}
    Anchors directly to the randomized trial target parameter:
    \begin{equation*}
        \theta_e = \mu, \quad \theta_o \mid \mu, \tau \sim \text{N}(\mu, 2\tau^2), \quad  k=\sqrt{2}\tau, \quad \tau \sim \text{HN}(s)
    \end{equation*}
    \end{block}

    Conditional on $\tau$, the shrinkage expectation for the trial parameter is:
    \begin{equation*}
        \E\{\theta_e \mid \hat\theta_e, \hat\theta_o, \tau\} = \frac{\sigma_e^{-2} \hat\theta_e + (\sigma_o^2 + 2\tau^2)^{-1} \hat\theta_o}{\sigma_e^{-2} + (\sigma_o^2 + 2\tau^2)^{-1}}.
    \end{equation*}

    \begin{itemize}
        \item Integrates over $\pi(\tau \mid \hat\theta_e, \hat\theta_o)$, fully propagating heterogeneity uncertainty.
        \item As conflict increases ($\tau^2 \to \infty$), observational borrowing vanishes.
    \end{itemize}
\end{frame}

%================================================
% SLIDE 8: Equivalence & Caveats
%================================================
\begin{frame}{Variance inflation vs. power priors and caveats}
    \begin{block}{Algebraic bridge between shrinkage and power priors}
    Equating observational shrinkage precision to power prior precision $a_0 \sigma_o^{-2}$:
    \begin{equation*}
        a_0 \sigma_o^{-2} = \frac{1}{\sigma_o^2 + 2\tau^2} \iff a_0 = \frac{\sigma_o^2}{\sigma_o^2 + 2\tau^2} = \frac{1}{1 + 2\frac{\tau^2}{\sigma_o^2}}
    \end{equation*}
    Hierarchical variance inflation by $2\tau^2$ is algebraically equivalent to power likelihood discounting (exponentiation) by $a_0 \in [0, 1]$.
    \end{block}

    \textbf{Limitations}:
    \begin{enumerate}
        \item \emph{Asymptotic normality}: Requires Gaussian summary estimates; fragile for rare-event binary outcomes.
        \item \emph{Dispersion vs. bias}: Symmetric offset $\text{N}(0, 2\tau^2)$ models random variation, not directional bias; observational data must be rigorously pre-adjusted.
        \item \emph{Aggregate evidence}: Operates strictly on contrast summaries, lacking patient-level covariate granularity and heterogeneity effect estimation.
    \end{enumerate}
\end{frame}

%================================================
% SLIDE 9: Power Likelihoods IPD
%================================================
\begin{frame}{Power likelihood approaches (Lin et al. (2026))}
    Lin et al. (2026) combine experimental trial ($X_e$) and observational cohort ($X_o$) at the individual participant level $X_i = (Y_i, T_i, W_i)$.

    \begin{block}{Joint power likelihood posterior (two-study setting)}
    With domain indicator $S_i \in \{e, o\}$ and $\eta \in [0, 1]$:
    \begin{equation*}
        \pi(\phi \mid X_e, X_o, \eta) \propto f_e(X_e; \phi) \{ f_o(X_o; \phi) \}^\eta \pi(\phi)
    \end{equation*}
    \end{block}

    \textit{Parameter partitioning}: \small $\phi = (\theta, \gamma, \psi) \in \R^p$, $p = p_\theta + p_\gamma + p_\psi$:
    \begin{itemize}
        \item $\theta \in \R^{p_\theta}$: Shared causal treatment effect contrast (parameter of interest).
        \item $\gamma \in \R^{p_\gamma}$: Trial-specific nuisance parameters.
        \item $\psi \in \R^{p_\psi}$: Observational nuisance parameters (confounding/selection).
    \end{itemize}

    \textbf{The role of $\eta$}: Discounting rate ($\eta = 0$: trial only posterior $\pi(\phi | X_e)$; $\eta = 1$: full unpenalized pooling) that accommodates both single-arm (control) and two-arm observational data.
\end{frame}

%================================================
% SLIDE 10: Frugal Parameterization
%================================================
\begin{frame}{Frugal parameterization and assumptions}

%Goal: Model non-causal features without imposing restrictive assumptions that might misspecify observational nuisance and bias the causal effect θ.

    \begin{block}{Frugal MSM factorisation (Evans \& Didelez, 2024)}
    Partition covariates $W = (C, Z)$ into effect modifiers $C$ and auxiliary $Z$:
    \begin{equation*}
        f_s(y, t, w) = f_s(t, z \mid c) f_s(c) \cdot f^*_{Y(t)\mid C}(y \mid t, c; \theta) \cdot c^*_s(y, t, z \mid c; \zeta_s)
    \end{equation*}
    \begin{itemize}
        \item $f^*$ models causal CATE $\tau(c)$; $\theta$ is variationally independent of nuisance $\zeta_e$;
        \item the copula $c^*_s$ links the outcome to $Z | C$ without feedback onto $\theta$.
    \end{itemize}
    \end{block}

    \textbf{Causal assumptions}:
    \begin{enumerate}
        \item \emph{Randomization}: Ensures trial ignorability and positivity, \small ($Y(t) \ci T \mid W, S = e$), $pr(T = 1|W,S = e) \in (0,1)$;
        \item \emph{Representativeness}: $\E\{Y(1) - Y(0) \mid C=c, S=e\} = \tau(c)$;
        \item \emph{Contrast transportability}: $\E\{Y(1) - Y(0) \mid C=c, S=o\} = \tau(c)$;
        \item \emph{Model specification}: Correct model specification for counterfactual $Y(t) \mid C$.
    \end{enumerate}
\end{frame}

%================================================
% SLIDE 11: Robust Selection of eta
%================================================
\begin{frame}{Robust selection of the influence factor $\eta$}
    \vspace{-0.25cm}
    \begin{block}{Cross-validation on experimental trial folds (Lin et al., 2026)}
    Fit on combined sample $\{D_e^{(-k)}, D_o\}$ with power $\eta$, evaluated on held-out trial units $D_e^{(k)}$:
    \begin{equation*}
        \widehat{\text{ELPD}}_{\text{cv}}(\eta) = \sum_{k=1}^K \sum_{i \in D_e^{(k)}} \log \int_{\R^p} f_e(X_i; \phi) \pi(\phi \mid X_e^{(-k)}, X_o, \eta) d\phi
    \end{equation*}
    Optimal grid tuning: $\hat\eta = \arg\max_{\eta \in \mathcal{H}} \widehat{\text{ELPD}}_{\text{cv}}(\eta)$ over a discrete grid $\mathcal{H} \subset [0, 1]$.
    \end{block}

    \textbf{Preserving trial internal validity}:
    \begin{itemize}
        \setlength{\itemsep}{0.04cm}
        \item \emph{RCT benchmark}: Evaluating CV strictly on $D_e$ preserves trial validity;
        \item \emph{Conflict penalty}: Residual confounding or non-transportability in $D_o$ degrades out-of-sample prediction on trial units, forcing $\hat\eta \to 0$;
        \item \emph{Efficiency gain}: Under causal concordance, variance reduction improves predictive density on held-out trial units, yielding $\hat\eta > 0$.
    \end{itemize}
\end{frame}

%================================================
% SLIDE 12: Asymptotics, Computation, and Inference
%================================================
\begin{frame}{Asymptotic approximations, computation, inference}
    \vspace{-0.15cm}
    \begin{block}{Bernstein--von Mises asymptotic decoupling}
    As $n_e, n_o \to \infty$, the joint posterior converges to a Gaussian distribution:
    \begin{equation*}
        \pi(\phi \mid X_e, X_o, \eta) \mathrel{\dot{\sim}} \text{N}\left(\hat\phi_\eta, \left(n_e V_e^{-1} + \eta n_o V_o^{-1}\right)^{-1}\right)
    \end{equation*}
    where $\hat\phi_\eta = (n_e V_e^{-1} + \eta n_o V_o^{-1})^{-1}(n_e V_e^{-1}\hat\phi_e + \eta n_o V_o^{-1}\hat\phi_o)$, with sandwich $V_e, V_o$.
    \end{block}

    \vspace{-0.12cm}
    \begin{itemize}
        \setlength{\itemsep}{0.04cm}
        \item \emph{MCMC-free}: Closed-form Gaussian density allows rapid grid search for $\hat\eta$, bypassing high-dimensional sampling over large registries.
        \item \emph{Privacy preservation}: Participating sites can share only summary estimates and sandwich matrices $(\hat\phi_s, V_s)$ %without pooling raw patient-level data.
        \item Given $\hat{\eta}$, the \emph{ATE} is evaluated via empirical g-computation over trial units:
        $\hat\tau^{(m)} = \frac{1}{n_e} \sum_{i \in D_e} [ \E\{Y(1) \mid C_i; \theta^{(m)}\} - \E\{Y(0) \mid C_i; \theta^{(m)}\} ]$, with $\theta^{(m)} \sim{\pi(\theta|X_e,X_o, \hat{\eta})}$.
        \item \textbf{Limitation:} Because $\hat{\eta}$ optimizes joint predictive density over all X rather than $\tau(c)$, moderate confounding may fail to trigger predictive penalties.
    \end{itemize}
    
\end{frame}

% SLIDE 13: Conclusions & Frontiers
%================================================
\begin{frame}{Remarks and research frontiers}
    \vspace{0.1cm}
    \textbf{Methodological evolution of Bayesian data fusion}:
    \begin{itemize}
        \setlength{\itemsep}{0.12cm}
        \item Transition from static subjective power priors to dynamic, data-adaptive borrowing.
        \item Pairwise reference shrinkage resolves the single-historical-study ($H=1$) bottleneck for contrast summaries.
        \item Combining power likelihoods with frugal parameterization unlocks granular, individual-level borrowing, protecting against observational misspecification.
    \end{itemize}

    \vspace{0.25cm}
    \textbf{What's next?}
    \begin{itemize}
        \setlength{\itemsep}{0.12cm}
        \item \emph{Contrast-targeted utility}: Formulating predictive scores focused strictly on the causal contrast $\tau(c)$ to eliminate transient under-coverage.
        \item \emph{Semiparametric copulas}: Relaxing parametric copula assumptions.
        \item \emph{Regulatory adoption}: Establishing formal guidelines for dynamic Bayesian borrowing in Phase III rare-disease and pediatric clinical trials.
    \end{itemize}
\end{frame}
