% SLIDE 3: Motivation
%================================================
\begin{frame}{Motivation: the data fusion dilemma}
    \vspace{0.05cm}
    \textbf{The clinical trade-off}:
    \begin{itemize}
        \setlength{\itemsep}{0.12cm}
        \item \emph{Randomised controlled trials (RCTs)}:
        \begin{itemize}
            \item Gold standard for internal validity through design ($Y(t) \ci T \mid W, S = e$).
            \item Small sample sizes, high costs, restrictive criteria: limited statistical power and poor generalisability.
        \end{itemize}
        \item \emph{Observational real-world data (RWD)}:
        \begin{itemize}
            \item Large patient cohorts, clinical registries, routine practice records.
            \item Vulnerable to unmeasured confounding and selection bias.
        \end{itemize}
    \end{itemize}

    \vspace{0.15cm}
    \begin{alertblock}{Objective of data fusion (Bareinboim \& Pearl, 2016)}
        Borrow statistical power from large observational cohorts while preserving the internal validity of the randomized trial.
    \end{alertblock}

    \vspace{0.1cm}
    The Bayesian paradigm offers a framework to introduce auxiliary evidence into prior distributions and adaptively discount it under data conflict.
\end{frame}

%================================================
% SLIDE 4: Baseline Power Prior
%================================================
\begin{frame}{Baseline: the power prior (Ibrahim \& Chen, 2000)}
    \vspace{0.1cm}
    Let $D_0 = (n_0, y_0, X_0)$ denote historical data with likelihood $L(\theta \mid D_0)$, and $\pi_0(\theta \mid c_0)$ an initial prior indexed by $c_0$.

    \begin{block}{Power prior and posterior formulation}
    For a discounting hyperparameter $a_0 \in [0, 1]$:
    \begin{equation*}
        \pi(\theta \mid D_0, a_0) \propto L(\theta \mid D_0)^{a_0} \pi_0(\theta \mid c_0), \quad
        \pi(\theta \mid D_e, D_0, a_0) \propto L(\theta \mid D_e) \pi(\theta \mid D_0, a_0)
    \end{equation*}
    \end{block}

    \begin{itemize}
        \item $a_0 = 1$ corresponds to unpenalized pooling; $a_0 = 0$ excludes external data.
        \item Disperses prior concentration around historical estimates to mitigate conflict.
    \end{itemize}

    \vspace{0.15cm}
    \textbf{Core limitations}:
    \begin{itemize}
        \setlength{\itemsep}{0.1cm}
        \item \emph{Fixed $a_0$}: Purely subjective elicitation; no internal commensurability check.
        \item \emph{Random $a_0$}: Naive joint prior $\pi(\theta, a_0 \mid D_0) \propto L^{a_0} \pi_0 \pi(a_0)$ violates the likelihood principle (Duan et al., 2006).
    \end{itemize}
\end{frame}

%================================================
% SLIDE 5: Modified Power Prior
%================================================
\begin{frame}{The modified power prior and the $H=1$ setting}
    \begin{block}{Proper power prior (Duan et al., 2006; Neuenschwander et al., 2009)}
    Retaining $C(a_0) = [ \int_\Theta L(\theta \mid D_0)^{a_0} \pi_0(\theta \mid c_0) d\theta ]^{-1}$ restores probability calculus:
    \begin{equation*}
        \pi(\theta, a_0 \mid D_0) = C(a_0) L(\theta \mid D_0)^{a_0} \pi_0(\theta \mid c_0) \pi(a_0) \quad (a_0 \in A)
    \end{equation*}
    \end{block}

    \begin{itemize}
        \item \emph{Prevents collapse}: Without $C(a_0)$, $\pi(a_0 \mid \text{data})$ collapses to 0 even for identical datasets.
        \item \emph{Dynamic commensurability}: Concordance pushes $a_0 \to 1$; conflict forces $a_0 \to 0$.
        \item \emph{Multi-study extension}: Assigns vector weights $a_0 = (a_{01}, \dots, a_{0H})^\T$ for $H \ge 2$.
    \end{itemize}

    \begin{highlightbox}
        \textbf{\textcolor{MyRed}{\small The single-source bottleneck ($H = 1$)}}\par\vspace{2pt}
        {\footnotesize With only one historical study ($H=1$), data cannot identify cross-study heterogeneity: $\pi(a_0 \mid \text{data})$ remains sensitive to its prior.}
    \end{highlightbox}
\end{frame}

%================================================
% SLIDE 6: Robust MAP Priors
%================================================
\begin{frame}{Robust meta-analytic-predictive (MAP) priors}
    \vspace{-0.15cm}
    \textbf{Schmidli et al. (2014)}: Shift to hierarchical predictive borrowing for the control response parameter $\theta = g(\psi)$ across $H \ge 3$ historical studies.

    \begin{block}{Hierarchical exchangeability model}
    \begin{equation*}
        \theta_1, \dots, \theta_H, \theta_* \mid \mu, \tau \overset{\text{i.i.d.}}{\sim} \text{N}(\mu, \tau^2), \quad \pi(\mu, \tau) = \pi(\mu)\pi(\tau)
    \end{equation*}
    where $\tau \ge 0$ quantifies between-trial heterogeneity.
    \end{block}

    \begin{itemize}
        \item \emph{Conjugate mixture}: Predictive density $\pi(\psi_* \mid \text{data}) \approx \sum_{k=1}^K w_k \pi_c(\psi_* \mid \xi_k)$ via EM algorithm provides closed-form posterior updates during trial analysis.
        \item \emph{Robustification}: Augment mixture with a vague component $\pi_V(\psi_*)$ weighted by $w_R \in [0.10, 0.20]$ (Cromwell's rule):
        \begin{equation*}
            \hat\pi_{\text{HR}}(\psi_*) = (1 - w_R) \sum_{k=1}^K w_k \pi_c(\psi_* \mid \xi_k) + w_R \pi_V(\psi_*)
        \end{equation*}
        \item \emph{Adaptive discounting}: Under prior--data conflict $(m_k \ll m_V)$, the Bayes factor ratio drives $\tilde{w}_R \to 1$, automatically discarding external controls.
    \end{itemize}
\end{frame}

%================================================
% SLIDE 7: Pairwise Shrinkage for H=1
%================================================
\begin{frame}{Pairwise shrinkage for $H = 1$ (R\"over \& Friede, 2020)}
    \begin{itemize}
        \item \emph{Pairwise synthesis} ($H=1$): Frequentist random-effects often yield $\hat\tau^2 = 0$, underestimating variance uncertainty.
        \item Symmetric exchangeability around an artificial mean $\mu$ is hard to justify.
    \end{itemize}

    \begin{block}{Asymmetric reference model ($\theta_e = \mu$)}
    Anchors directly to the randomized trial target parameter:
    \begin{equation*}
        \theta_e = \mu, \quad \theta_o \mid \mu, \tau \sim \text{N}(\mu, 2\tau^2), \quad  k=\sqrt{2}\tau, \quad \tau \sim \text{HN}(s)
    \end{equation*}
    \end{block}

    Conditional on $\tau$, the shrinkage expectation for the trial parameter is
    \begin{equation*}
        \E\{\theta_e \mid \hat\theta_e, \hat\theta_o, \tau\} = \frac{\sigma_e^{-2} \hat\theta_e + (\sigma_o^2 + 2\tau^2)^{-1} \hat\theta_o}{\sigma_e^{-2} + (\sigma_o^2 + 2\tau^2)^{-1}}
    \end{equation*}

    \begin{itemize}
        \item Integrates over $\pi(\tau \mid \hat\theta_e, \hat\theta_o)$, fully propagating heterogeneity uncertainty.
        \item As conflict increases ($\tau^2 \to \infty$), observational borrowing vanishes.
    \end{itemize}
\end{frame}

%================================================
% SLIDE 8: Equivalence & Caveats
%================================================
\begin{frame}{Variance inflation vs. power priors and caveats}
    \begin{block}{Analytical equivalence (R\"over \& Friede, 2020)}
    Equating observational shrinkage precision to power prior precision $a_0 \sigma_o^{-2}$:
    \begin{equation*}
        a_0 \sigma_o^{-2} = \frac{1}{\sigma_o^2 + 2\tau^2} \iff a_0 = \frac{\sigma_o^2}{\sigma_o^2 + 2\tau^2} = \frac{1}{1 + 2\frac{\tau^2}{\sigma_o^2}}
    \end{equation*}
    Hierarchical variance inflation by $2\tau^2$ is algebraically equivalent to power likelihood discounting by $a_0 \in [0, 1]$.
    \end{block}

    \textbf{Methodological caveats}:
    \begin{enumerate}
        \item \emph{Asymptotic normality}: Requires Gaussian summary estimates; fragile for rare-event binary outcomes.
        \item \emph{Dispersion vs. bias}: Symmetric offset $\text{N}(0, 2\tau^2)$ models random variation, not directional bias; observational data must be rigorously pre-adjusted.
        \item \emph{Aggregate evidence}: Operates strictly on contrast summaries, lacking patient-level covariate granularity.
    \end{enumerate}
\end{frame}

%================================================
% SLIDE 9: Power Likelihoods IPD
%================================================
\begin{frame}{Individual-level fusion: power likelihoods}
    \textbf{Lin et al. (2026)}: Combine experimental trial ($X_e$) and observational cohort $X_o$ at the individual participant level $X_i = (Y_i, T_i, W_i)$.

    \begin{block}{Joint power likelihood}
    With domain indicator $S_i \in \{e, o\}$ and discounting rate $\eta \in [0, 1]$:
    \begin{equation*}
        \pi(\phi \mid X_e, X_o, \eta) \propto f_e(X_e; \phi) \{ f_o(X_o; \phi) \}^\eta \pi(\phi)
    \end{equation*}
    \end{block}

    \textbf{Structured parameter partitioning}: $\phi = (\theta, \gamma, \psi) \in \R^p$ ($p = p_\theta + p_\gamma + p_\psi$):
    \begin{itemize}
        \item $\theta \in \R^{p_\theta}$: Shared causal treatment effect contrast (parameter of interest).
        \item $\gamma \in \R^{p_\gamma}$: Trial-specific nuisance parameters.
        \item $\psi \in \R^{p_\psi}$: Observational nuisance parameters (confounding/selection).
    \end{itemize}

    \emph{Flexibility}: Continuous discounting ($\eta = 0$: trial only; $\eta = 1$: full pooling) accommodates both single-arm (control) and two-arm observational data.
\end{frame}

%================================================
% SLIDE 10: Frugal Parameterization
%================================================
\begin{frame}{Frugal parameterization and identification}

%Goal: Model non-causal features without imposing restrictive assumptions that might misspecify observational nuisance and bias the causal effect θ.

    \begin{block}{Frugal MSM factorisation (Evans \& Didelez, 2024)}
    Partition covariates $W = (C, Z)$ into effect modifiers $C$ and auxiliary $Z$:
    \begin{equation*}
        f_s(y, t, w) = f_s(t, z \mid c) f_s(c) \cdot f^*_{Y(t)\mid C}(y \mid t, c; \theta) \cdot c^*_s(y, t, z \mid c; \zeta_s)
    \end{equation*}
    $f^*$ models causal CATE $\tau(c)$; copula $c^*_s$ absorbs domain nuisance ($\zeta_e = \gamma, \zeta_o = \psi$) without feedback onto $\theta$.
    \end{block}

    \textbf{Causal identification pillars}:
    \begin{enumerate}
        \item \emph{Randomization}: Ensures trial ignorability and positivity ($Y(t) \ci T \mid W, S = e$).
        \item \emph{Representativeness}: $\E\{Y(1) - Y(0) \mid C=c, S=e\} = \tau(c)$.
        \item \emph{Contrast transportability}: $\E\{Y(1) - Y(0) \mid C=c, S=o\} = \tau(c)$.
        \item \emph{Model specification}: Correct causal MSM for counterfactuals $Y(t) \mid C$.
    \end{enumerate}
\end{frame}

%================================================
% SLIDE 11: Adaptive Tuning & Asymptotics
%================================================
\begin{frame}{Adaptive tuning, asymptotics, and privacy}
    \textbf{Adaptive selection}: $\hat\eta = \arg\max_{\eta \in \mathcal{H}} \widehat{\text{ELPD}}_{\text{cv}}(\eta)$ evaluated on held-out RCT folds $D_e^{(k)}$.
    It preserves trial internal validity: unmeasured residual confounding forces $\hat\eta \to 0$.

    \begin{block}{Bernstein--von Mises asymptotic decoupling}
    As $n_e, n_o \to \infty$, the joint posterior converges to a Gaussian distribution:
    \begin{equation*}
        \pi(\phi \mid X_e, X_o, \eta) \mathrel{\dot{\sim}} \text{N}\left(\hat\phi_\eta, \left(n_e V_e^{-1} + \eta n_o V_o^{-1}\right)^{-1}\right)
    \end{equation*}
    with precision-weighted mean $\hat\phi_\eta = (n_e V_e^{-1} + \eta n_o V_o^{-1})^{-1}(n_e V_e^{-1}\hat\phi_e + \eta n_o V_o^{-1}\hat\phi_o)$, with sandwich $V_e, V_o$.
    \end{block}

    \begin{itemize}
        \item \emph{MCMC-free optimization}: Closed-form Gaussian density allows rapid grid search.
        \item \emph{Privacy preservation}: Participating sites share only summaries $(\hat\phi_s, V_s)$ without pooling raw individual patient data.
    \end{itemize}

    %Target ATE ˆτ obtained via empirical g-computation.
\end{frame}

% SLIDE 13: Conclusions & Frontiers
%================================================
\begin{frame}{Concluding remarks and research frontiers}
    \vspace{0.1cm}
    \textbf{Summary of methodological evolution}:
    \begin{itemize}
        \setlength{\itemsep}{0.12cm}
        \item Transition from static subjective power priors to dynamic, data-adaptive borrowing.
        \item Pairwise reference shrinkage resolves the single-historical-study ($H=1$) bottleneck for contrast summaries.
        \item Combining power likelihoods with frugal parameterization unlocks granular, individual-level borrowing while protecting against observational misspecification.
    \end{itemize}

    \vspace{0.25cm}
    \textbf{Open research frontiers}:
    \begin{itemize}
        \setlength{\itemsep}{0.12cm}
        \item \emph{Contrast-targeted utility}: Formulating predictive scores focused strictly on the causal contrast $\tau(c)$ to eliminate transient under-coverage.
        \item \emph{Semiparametric copulas}: Relaxing parametric copula assumptions.
        \item \emph{Regulatory adoption}: Establishing formal guidelines for dynamic Bayesian borrowing in Phase III rare-disease and pediatric clinical trials.
    \end{itemize}
\end{frame}
